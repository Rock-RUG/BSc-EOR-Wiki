\documentclass[tikz,border=2pt]{standalone}

\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{groupplots}

\begin{document}
	
	\begin{tikzpicture} 
		\begin{groupplot}[ 
			group style={ 
				group size=2 by 1, 
				horizontal sep=1cm, 
			}, 
			domain=-1.5:1.5, 
			ymin=-1.5, ymax=1.5, 
			legend pos=south east, 
			width=0.5\linewidth, 
			] 
			
			\nextgroupplot[] 
			\addplot[blue, thick] {sin(deg(x))}; 
			\addlegendentry{$\sin x$} 
			\addplot[red, thick] {x}; 
			\addlegendentry{$x$} 
			
			\nextgroupplot[] 
			\addplot[blue, thick] {sin(deg(x))}; 
			\addlegendentry{$\sin x$} 
			\addplot[brown, thick] {x - x^3/6}; 
			\addlegendentry{$x - x^3/6$} 
			
		\end{groupplot} 
	\end{tikzpicture} 
	
\end{document}
