		0 & |[fill=blue!25]| 1 & 0 \\
		0 & 0 & |[fill=blue!25]| 1 \\
	};
	\node[anchor=south] at (I.north) {$I_3$};
	\begin{scope}[xshift=3.2cm, yshift=-0.6cm]
		\draw[->] (-0.4,0) -- (2.4,0);

	\end{scope}
