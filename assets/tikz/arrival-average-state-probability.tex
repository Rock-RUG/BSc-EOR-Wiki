\begin{tikzpicture}[font=\small, >={Latex[length=2mm]}, x=0.6cm, y=0.9cm]

  \foreach \x in {0,10} \draw (\x,0) -- (\x,-0.08) node[anchor=north] {$\x$};
  \foreach \y in {1,2} {
    \draw (0,\y) -- (-0.12,\y) node[anchor=east] {$\y$};
    \draw[gray!50, thin, dashed] (0,\y) -- (10,\y);
