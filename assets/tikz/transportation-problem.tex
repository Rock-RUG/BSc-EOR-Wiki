	\node[v] (p1) at (0,1.5) {$P_1$}; \node[v] (p2) at (0,-1.5) {$P_2$};

	\node[blue!70!black, font=\small, left=4pt of p2] {supply $20$};

	\node[red!70!black, font=\small, right=4pt of c2] {demand $20$};
