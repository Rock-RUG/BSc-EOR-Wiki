% Entry game, payoffs (entrant, incumbent): Out (0,2); after In, A (1,1) or F (-1,-1).
% Left: the Nash equilibrium (Out, F). Its path is Out alone; the incumbent's node is off the path,
% and the planned F there (orange) is never carried out. Right: the SPNE (In, A) on the same tree;
% now the incumbent's node is on the path In, A. On or off the path is a property of the profile.
\begin{tikzpicture}[font=\small, >={Latex[length=2mm]},
    dec/.style={circle, fill, inner sep=1.6pt}]
  \foreach \panel in {0,1} {
  \begin{scope}[xshift=\panel*5.3cm]
    \node[dec, blue!70!black, label={[blue!70!black]above:entrant}] (r) at (1.4,0) {};
    \node[anchor=north, inner sep=2pt] at (0,-1.45) {$(0,2)$};
    \node[anchor=north, inner sep=2pt] at (2.0,-2.75) {$(1,1)$};
    \node[anchor=north, inner sep=2pt] at (3.6,-2.75) {$(-1,-1)$};
  \end{scope}
  }
  % ---- left: (Out, F)
  \draw[very thick, green!50!black] (1.4,0) -- node[above left, pos=0.5] {Out} (0,-1.4);
  \draw[thin, gray] (1.4,0) -- node[above right, pos=0.5] {In} (2.8,-1.4);
  \draw[thin, gray] (2.8,-1.4) -- node[left, pos=0.55] {$A$} (2.0,-2.7);
  \draw[thick, orange!90!black, dashed] (2.8,-1.4) -- node[right, pos=0.55] {$F$} (3.6,-2.7);
  \draw[red!70!black, thick, fill=white] (2.8,-1.4) circle (2.2pt);
  \node[red!70!black, anchor=west, align=left] at (2.9,-1.3) {incumbent:\\off path};
  \node[anchor=north] at (1.8,-3.45) {NE $(\text{Out},F)$};
  % ---- right: (In, A)
  \begin{scope}[xshift=5.3cm]
    \draw[thin, gray] (1.4,0) -- node[above left, pos=0.5] {Out} (0,-1.4);
    \draw[very thick, green!50!black] (1.4,0) -- node[above right, pos=0.5] {In} (2.8,-1.4);
    \draw[very thick, green!50!black] (2.8,-1.4) -- node[left, pos=0.55] {$A$} (2.0,-2.7);
    \draw[thin, gray] (2.8,-1.4) -- node[right, pos=0.55] {$F$} (3.6,-2.7);
    \node[dec, red!70!black] at (2.8,-1.4) {};
    \node[red!70!black, anchor=west, align=left] at (2.9,-1.3) {incumbent:\\on path};
    \node[anchor=north] at (1.8,-3.45) {SPNE $(\text{In},A)$};
  \end{scope}
\end{tikzpicture}
\end{document}
