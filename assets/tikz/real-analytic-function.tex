\documentclass[tikz,border=2pt]{standalone}

\usepackage{pgfplots}
\usepackage{mathtools}
\pgfplotsset{compat=1.18}

\begin{document}
	
	\begin{tikzpicture}
		\begin{axis}[
			domain=-0.5:1,
			samples=200,
			xlabel={$x$},
			ylabel={$f(x)$},
			axis lines=middle,
			xtick={0},
			ytick={0},
			ymin=-0.2,
			ymax=1,
			enlarge x limits=false,
			enlarge y limits=false,
			]
			
			% ===== 分段函数文字 =====
			\node at (axis description cs:0.672,0.73) {$
				f(x)=
				\begin{dcases}
					\,e^{-1/x}, & \text{for } x>0,\\
					\,0, & \text{for } x\leq 0.
				\end{dcases}
				$};
			
			% ===== 图像 =====
			\addplot[
			domain=0.001:1,
			thick,
			] {exp(-1/x)};
			
			\addplot[
			domain=-0.5:0,
			thick,
			] {0};
			
		\end{axis}
	\end{tikzpicture}
	
\end{document}
