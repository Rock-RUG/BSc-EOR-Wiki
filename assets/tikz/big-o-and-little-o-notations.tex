\documentclass[tikz,border=2pt]{standalone}

\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\begin{document}
	
	\begin{tikzpicture}
		\begin{axis}[ 
			xmin=-0.5,xmax=1, 
			ymin=-1,ymax=1, 
			axis lines=middle, 
			xlabel=$x$, 
			ylabel=$y$, 
			legend pos=south east, 
			legend cell align={left} 
			] 
			\addplot[thick,domain=-0.5:1, samples=100, color=purple]{x};

			\legend{$f(x)=x$,$g(x)=x^2$,$h(x)=2x^2+x$}
