  \foreach \k in {0,1,2,3} {
    \draw[thick] (\k,0.12) -- (\k,-0.12);
    \node[anchor=north] at (\k,-0.15) {$\k$};
  }
  \node[anchor=north] at (2.25,-0.15) {$2.25$};
  % death
  \node[blue!70!black, font=\small\bfseries] at (2.13,0) {$\times$};
  \node[blue!70!black, anchor=south east] at (2.1,0.12) {$T_x=2.13$};
  % payments
  \draw[->, red, very thick] (2.25,-0.62) -- (2.25,-1.5);
  \node[red!70!black, anchor=north] at (2.25,-1.55) {$m=4$};
  \draw[->, red!60, very thick, dashed] (3,-0.62) -- (3,-1.5);
  \node[red!70!black, anchor=north] at (3,-1.55) {$m=1$};
  % waiting time
  \draw[green!50!black, thick, decorate, decoration={brace, amplitude=3pt}]
    (2.13,0.25) -- (2.25,0.25);
