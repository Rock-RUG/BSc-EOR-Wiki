    \node[blue!70!black, anchor=west] at (axis cs:1.08,1) {$X$};
    \node[red!70!black, anchor=west] at (axis cs:2.08,0.5) {$Y$};

    \node[blue!70!black, anchor=north east] at (axis cs:0.5,0.48) {$X$};

  \end{axis}
