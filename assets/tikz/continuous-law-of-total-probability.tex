		]
		\addplot[blue!55, thick] coordinates {(0,4) (0.25,4) (0.25,0)};

		\addplot[blue!55, thick] coordinates {(0,1.333) (0.75,1.333) (0.75,0)};

		\addplot[red, very thick, domain=0:1] {2*(1-x)};

	\end{axis}
