\documentclass[tikz,border=2pt]{standalone}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\begin{document}
% A finite sequence has a last term: (a_i)_{i=0}^{5} lists six values in order and then stops.
\begin{tikzpicture}
	\begin{axis}[
		axis lines=middle,
		xmin=-0.6, xmax=7.2,
		ymin=0, ymax=3.6,
		xtick={0,...,5}, ytick=\empty,
		xlabel={$i$}, ylabel={$a_i$},
		xlabel style={below right}, ylabel style={above left},
		clip=false,
		width=8cm, height=4.8cm
		]
		\addplot[only marks, mark=*, mark size=2pt, blue] coordinates {(0,1.2) (1,2.1) (2,1.6) (3,2.9) (4,2.4) (5,3.1)};
		\draw[blue!40] (axis cs:0,0) -- (axis cs:0,1.2);
		\draw[blue!40] (axis cs:1,0) -- (axis cs:1,2.1);
		\draw[blue!40] (axis cs:2,0) -- (axis cs:2,1.6);
		\draw[blue!40] (axis cs:3,0) -- (axis cs:3,2.9);
		\draw[blue!40] (axis cs:4,0) -- (axis cs:4,2.4);
		\draw[blue!40] (axis cs:5,0) -- (axis cs:5,3.1);
		\node[blue, anchor=south, font=\small] at (axis cs:0,1.3) {$a_0$};
		\node[blue, anchor=south, font=\small] at (axis cs:5,3.2) {$a_5$ (last term)};
		\node[gray, anchor=west, font=\small] at (axis cs:5.6,1.5) {no $a_6$};
	\end{axis}
\end{tikzpicture}
\end{document}
