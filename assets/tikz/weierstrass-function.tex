\documentclass[tikz,border=2pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\begin{document}
% Partial sum f_8(x) = sum_{n=1}^{8} a^n cos(b^n x) with a = 1/2, b = 3 (so ab > 1), on [0, 1]
% (left) and zoomed into [0.45, 0.55] (right): the zoom shows the same kind of irregularity.
\begin{tikzpicture}[every node/.style={font=\small},
	declare function={W(\x)=0.5*cos(deg(3*\x))+0.25*cos(deg(9*\x))+0.125*cos(deg(27*\x))+0.0625*cos(deg(81*\x))+0.03125*cos(deg(243*\x))+0.015625*cos(deg(729*\x))+0.0078125*cos(deg(2187*\x))+0.00390625*cos(deg(6561*\x));}]
	\pgfplotsset{wf/.style={
		width=6.2cm, height=4.8cm,
		ytick=\empty, tick label style={font=\small},
		axis line style={gray!70},
		xlabel={$x$}, xlabel style={font=\small},
		no markers, clip=true}}
	\begin{axis}[wf, name=left, domain=0:1, samples=2000, xmin=0, xmax=1, ymin=-1.05, ymax=1.05, xtick={0,0.5,1}]
		\addplot[blue, thin] {W(x)};
		\draw[red, thick] (axis cs:0.45,-1) rectangle (axis cs:0.55,1);
	\end{axis}
	\begin{axis}[wf, at={(left.east)}, anchor=west, xshift=0.9cm, domain=0.45:0.55, samples=2000, xmin=0.45, xmax=0.55, xtick={0.45,0.5,0.55}, xticklabel style={/pgf/number format/fixed, /pgf/number format/precision=2}, axis line style={red, thick}]
		\addplot[blue, thin] {W(x)};
	\end{axis}
\end{tikzpicture}
\end{document}
