	\begin{axis}[

		ymin=0, ymax=5.2,
		xtick={1,5,10,15}, extra x ticks={7}, extra x tick labels={$N$}, extra x tick style={tick label style={red}},

		xlabel={$n$}, ylabel={$a_n$},

		\node[blue, anchor=north west, font=\small] at (axis cs:1,5.0) {$a_n\to+\infty$};
