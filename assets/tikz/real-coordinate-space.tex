\documentclass[tikz,border=2pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{tikz}
\usetikzlibrary{arrows.meta}

\begin{document}
% R^3: a point is an ordered triple (x1, x2, x3), its three coordinates measured along the axes.
\begin{tikzpicture}[scale=1.0, every node/.style={font=\small}, >={Latex[length=2.2mm]}]
	\draw[->] (0,0) -- (3.6,0) node[right] {$x_1$};
	\draw[->] (0,0) -- (0,3.0) node[above] {$x_3$};
	\draw[->] (0,0) -- (-1.9,-1.6) node[below left] {$x_2$};
    % Projected coordinate basis: e1=(1,0), e2=(-0.76,-0.64), e3=(0,1).
    % The tuple is (3,1,2.3); every foot and edge comes from the same basis.
    \def\coordone{3}\def\coordtwo{1}\def\coordthree{2.3}
    \def\eTwoX{-0.76}\def\eTwoY{-0.64}
    \coordinate (Xone) at (\coordone,0);
    \coordinate (Xtwo) at ({\coordtwo*\eTwoX},{\coordtwo*\eTwoY});
    \coordinate (Xthree) at (0,\coordthree);
    \coordinate (Pxy) at ({\coordone+\coordtwo*\eTwoX},{\coordtwo*\eTwoY});
    \coordinate (P) at ({\coordone+\coordtwo*\eTwoX},{\coordtwo*\eTwoY+\coordthree});
    \draw[dashed, gray] (Xone) -- (Pxy) -- (Xtwo);
    \draw[dashed, gray] (Pxy) -- (P) -- (Xthree);
    \fill[red] (P) circle (2.2pt) node[above right] {$\boldsymbol{x}=(x_1,x_2,x_3)$};
    \node[below right, inner sep=2pt] at (Xone) {$x_1$};
    \node[left] at (Xthree) {$x_3$};
    \node[above left, inner sep=2pt] at (Xtwo) {$x_2$};
	\node[anchor=north] at (0.9,-1.95) {$\mathbb{R}^n=\{(x_1,\dots,x_n)\colon x_i\in\mathbb{R}\}$; here $n=3$};
\end{tikzpicture}
\end{document}
